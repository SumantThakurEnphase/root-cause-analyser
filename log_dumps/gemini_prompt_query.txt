- CDN serving old assets after a deployment.

function getWeightedTotalSolarResourceFraction(roofPlaneShadingData) {
	let totalSolarResourceFraction = 0;
	let totalPanelCount = 0;
	roofPlaneShadingData.forEach((planeShadingData) => {
		totalSolarResourceFraction += planeShadingData.panelShading.tsrfYearly * planeShadingData.count;
		totalPanelCount += planeShadingData.count;
	});
	totalSolarResourceFraction /= totalPanelCount;
	return totalSolarResourceFraction;
}

			selectedAnalysis,
			projectConnection,
			solarRate,
			financialOption.rate_escalator,
			firstYearMonthlyPayment
		);
		return {
			firstYearEstimatedProduction: firstYearProduction,
			systemSize: {
				unit: 'kW',
				value: systemSize,
			},
			offerId: financialOption.financing_partner_metadata.offerId,
			batteryCost: {
				type: 'USD',
				value: proposal.storagePrice,
